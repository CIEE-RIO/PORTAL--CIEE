rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function autenticado() {
      return request.auth != null && request.auth.token.email != null;
    }

    function perfilUsuario() {
      return get(/databases/$(database)/documents/usuarios_login/$(request.auth.token.email));
    }

    function admin() {
      return autenticado()
        && exists(/databases/$(database)/documents/usuarios_login/$(request.auth.token.email))
        && perfilUsuario().data.perfil == "admin";
    }

    function podeEscrever() {
      return autenticado()
        && exists(/databases/$(database)/documents/usuarios_login/$(request.auth.token.email))
        && (perfilUsuario().data.perfil == "admin"
          || perfilUsuario().data.podeEscrever == true);
    }

    match /usuarios_login/{document=**} {
      allow read: if autenticado();
      allow write: if admin();
    }

    match /{document=**} {
      allow read: if autenticado();
      allow write: if podeEscrever();
    }
  }
}
